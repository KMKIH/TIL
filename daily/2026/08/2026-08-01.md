    1. BruteForce
        ```cpp
            class Solution {
            private:
                vector<int> ret;
                bool isSatisfied(vector<int>& arr){
                    for(int i = 0;i<arr.size();i++){
                        for(int j = 0;j<arr.size();j++){
                            if(!(arr[i]%arr[j] == 0 || arr[j]%arr[i] == 0))return false;
                        }
                    }
                    return true;
                }
            public:
                void function(vector<int>& nums, vector<int>& cur, int idx){
                    if(isSatisfied(cur) == false) return;
                    if(ret.size() < cur.size()) ret = cur;
                    for(int i = idx + 1; i < nums.size();i++){
                        cur.push_back(nums[i]);
                        function(nums, cur, i);
                        cur.pop_back();
                    }
                }
                vector<int> largestDivisibleSubset(vector<int>& nums) {
                    vector<int> arr;
                    function(nums,arr,-1);
                    return ret;
                }
            };
        ```

    2. 최적화1 
        - isSatisfied() 함수에서 계산했던값을 지속적으로 계산하고 있다. 한 원소를 추가할때마다 모든 원소에 대해서 비교하는게 아니라, 새로운 원소와 기존 원소들의 비교로 변경해보자
        ```cpp
            class Solution {
            private:
                vector<int> ret;
                bool isSatisfied(vector<int>& arr, int newElement){
                    for(int i = 0;i<arr.size();i++){
                        if(!(arr[i]%newElement == 0 || newElement%arr[i] == 0))return false;
                    }
                    return true;
                }
            public:
                void function(vector<int>& nums, vector<int>& cur, int idx){
                    if(ret.size() < cur.size()) ret = cur;
                    for(int i = idx + 1; i < nums.size();i++){
                        if(isSatisfied(cur, nums[i])){
                            cur.push_back(nums[i]);
                            function(nums, cur, i);
                            cur.pop_back();
                        }
                    }
                }
                vector<int> largestDivisibleSubset(vector<int>& nums) {
                    vector<int> arr;
                    function(nums,arr,-1);
                    return ret;
                }
            };
        ```
        **=> 아직도 시간초과**
    3. 최적화2
        - 어차피 하나에 대해서 비교할거면, 미리 계산해놔도 되지않나?
        ```cpp
        class Solution {
        private:
            vector<int> ret;
            int isSatisfiedArr[1001][1001];
            bool isSatisfied(vector<int>& arr, int newIdx){
                for(int i = 0;i<arr.size();i++){
                    if(isSatisfiedArr[arr[i]][newIdx] == false) return false;
                }
                return true;
            }
        public:
            void function(vector<int>& nums, vector<int>& cur, int idx){
                if(ret.size() < cur.size()) ret = cur;
                for(int i = idx + 1; i < nums.size();i++){
                    if(isSatisfied(cur, i)){
                        cur.push_back(i);
                        function(nums, cur, i);
                        cur.pop_back();
                    }
                }
            }
            vector<int> largestDivisibleSubset(vector<int>& nums) {
                for(int i = 0;i<nums.size();i++){
                    for(int j = 0;j<nums.size();j++){
                        isSatisfiedArr[i][j] = nums[i]%nums[j] == 0 || nums[j]%nums[i] == 0;
                    }
                }

                vector<int> arr;
                function(nums,arr,-1);
                for(int i = 0;i<ret.size();i++){
                    ret[i] = nums[ret[i]];
                }
                return ret;
            }
        };
        ```
        **=> 아직도 시간초과**
    3. 모르겠어서 AI에게 힌트를 받았다...
        > 부분집합을 직접 만드는 방식에서 벗어나야 한다.
        > 정렬 이후, a < b라면 b % a == 0만 확인하면 된다.
        - % 연산에 대해 생각을 안하고 코드를 작성했다... % 연산은 '나누어 떨어지는가'에 대한 연산이므로, 이미 앞에서 나누어떨어지면, 그 뒤는 계산할 필요가 없다!
        - 그러면 이제 문제는 간단해진다. nums[i]의 부분 집합의 갯수를 구하자!
        ```cpp
        dp[i] = nums[i]를 마지막 원소로 사용하는 부분집합의 최대 갯수
        ```
    4. 1단계: DP로 구성
        ```cpp
        for(int i = 1;i<nums.size();i++){
            for(int j = 0;j<i;j++){
                if(nums[i] % nums[j] == 0 && res < dp[j] + 1){
                    dp[i] = dp[j] + 1;
                }
            }
        }
        ```
