## 정리
### 코딩 테스트
- Rotting Oranges
    - 문제 접근
        - 바로 든 생각
            - BFS, queue 2개 사용(q,nq) 레이어별로 접근하여 썩는데 얼마나 걸리는지 파악, BFS 종료 이후 썩지 않은 오랜지가 있다면 -1 반환
    - 구현
        ```cpp
        class Solution {
        private:
            int dx[4] = {-1,0,0,1};
            int dy[4] = {0,-1,1,0};
        public:
            int orangesRotting(vector<vector<int>>& grid) {
                queue<pair<int,int>> q;
                queue<pair<int,int>> nq;

                int n = grid.size();
                int m = grid[0].size();

                int count = 0;
                for(int i = 0;i<n;i++){
                    for(int j = 0;j<m;j++){
                        if(grid[i][j] == 2) q.push({i,j});
                        if(grid[i][j] == 1) count++;
                    }
                }
                // 예외처리 - 멀쩡한 오랜지가 없다
                if(count == 0) return 0;

                int ret = 0;
                while(q.empty() == false){
                    while(q.empty() == false){
                        auto cur = q.front();q.pop();
                        int cx = cur.first;
                        int cy = cur.second;

                        for(int i = 0;i<4;i++){
                            int nx = cx + dx[i];
                            int ny = cy + dy[i];

                            if(nx < 0 || nx>=n || ny < 0 || ny >= m) continue;
                            if(grid[nx][ny] == 1){
                                grid[nx][ny] = 2;
                                count--;
                                nq.push({nx,ny});
                            }
                        }
                    }
                    q = queue<pair<int,int>>(nq);
                    nq = queue<pair<int,int>>();
                    ret++;
                }

                if(count > 0)return -1;
                return ret-1;
            }
        };
        ```
