## 정리
### 코딩 테스트
- Longest Increasing Path in a Matrix
    - **풀이1**
        > DP
        - 각 위치에서 DFS를 사용하여 최장 증가 경로를 저장한다
        - 이미 계산된 경로에 대해서는 dp값을 사용한다
        ```cpp
        class Solution {
        private:
            int dx[4] = {-1,0,0,1};
            int dy[4] = {0,-1,1,0};
            int n,m;
            vector<vector<int>> dp;

            int dfs(vector<vector<int>>& matrix, int x, int y){
                if(dp[x][y] != -1) return dp[x][y];
                dp[x][y] = 1;

                int res  = 0;
                int cur = matrix[x][y];
                for(int i = 0;i<4;i++){
                    int nx = x + dx[i];
                    int ny = y + dy[i];
                    if(nx < 0 || nx >= n || ny <0 || ny >= m) continue;
                    int next = matrix[nx][ny];
                    if(cur >= next) continue;
                    res = max(res, dfs(matrix, nx, ny));
                }
                return dp[x][y] = dp[x][y] + res;
            }
        public:
            int longestIncreasingPath(vector<vector<int>>& matrix) {
                n = matrix.size();
                m = matrix[0].size();
                dp= vector<vector<int>>(n,vector<int>(m,-1));

                int ret = 0;
                for(int i = 0; i < n; i++){
                    for(int j = 0;j<m;j++){
                        ret = max(ret, dfs(matrix, i,j));
                    }
                }
                return ret;
            }
        };
        ```
    - **풀이2**
        > 위상정렬
        - 위의 방법이 경로의 임의 지점에서 시작했다면, 경로의 시작지점부터 접근해보자
        - indegree가 0인 지점을 하나씩 살펴보자
        ```cpp
        class Solution {
        private:
            int dx[4] = {-1,0,0,1};
            int dy[4] = {0,-1,1,0};
        public:
            int longestIncreasingPath(vector<vector<int>>& matrix) {
                int n = matrix.size();
                int m = matrix[0].size();
                vector<vector<int>> indegree(n,vector<int>(m,0));

                for(int x = 0; x < n; x++){
                    for(int y = 0; y < m; y++){
                        int cx = x;
                        int cy = y;
                        int cur = matrix[cx][cy];
                        for(int i = 0;i<4;i++){
                            int nx = cx + dx[i];
                            int ny = cy + dy[i];
                            if(nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                            int next = matrix[nx][ny];
                            if(cur > next) indegree[cx][cy]++;
                        }
                    }
                }

                
                // 위상 정렬
                vector<vector<int>> dp(n,vector<int>(m,0));
                queue<pair<int,int>> q;
                queue<pair<int,int>> nq;
                for(int i = 0; i < n; i++){
                    for(int j = 0; j < m; j++){
                        if(indegree[i][j] == 0) {
                            q.push({i,j});
                            dp[i][j] = 1;
                        }
                    }
                }

                while(q.empty() == false){
                    while(q.empty() == false){
                        auto curPair = q.front();q.pop();
                        int cx = curPair.first;
                        int cy = curPair.second;
                        int cur = matrix[cx][cy];

                        for(int i = 0;i<4;i++){
                            int nx = cx + dx[i];
                            int ny = cy + dy[i];
                            if(nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                            int next = matrix[nx][ny];

                            if(cur < next){
                                dp[nx][ny] = dp[cx][cy] +1;
                                indegree[nx][ny]--;
                                if(indegree[nx][ny] == 0)
                                    q.push({nx,ny});
                            }
                        }
                    }
                    q = queue<pair<int,int>>(nq);
                    nq = queue<pair<int,int>>();
                }
                
                int ret = 0;
                for(int i = 0; i < n; i++){
                    for(int j = 0; j < m; j++){
                        ret = max(ret, dp[i][j]);
                    }
                }
                return ret;
            }
        };
        ```
