## 정리
### 코딩 테스트
- Maximal Square
    > 0. dp 부터 생각하지 말고 재귀부터 생각한다
    
    - 문제 접근
        - 1부터 min(n,m)까지 모든 사이즈에 대해서 사각형을 찾는다
    
    > 1. 상태 정의

    - dp[s][i][j] = (i,j)에서 size가 s인 사각형을 만들 수 있는가?

    > 4. 점화식

    - dp[s][i][j] = dp[s-1][i+1][j] && dp[s-1][i][j+1] && dp[s-1][i+1][j+1]

    > 5. 최적화

    1. 사이즈 별 dp 저장이 필요할까?
        - 좌상단 → 우하단 순으로 순회하면 현재 상태가 참조하는 오른쪽/아래쪽 값들은 아직 갱신되지 않았으므로 이전 size의 DP 값을 그대로 사용할 수 있다.

        => dp[s][i][j]

        => 결과 : 395ms / 57.1MB -> 296ms / 27.3MB
    
    2. 사이즈 별 접근이 필요한가?
        - 사이즈 별 접근 자체가 중복된 계산을 만든다
        - dp[i][j] = i,j를 우측 하단으로 하는 사각형의 크기
        - dp[i][j] = min({dp[i][j-1],dp[i-1][j],dp[i-1][j-1]})+1;

    ---
    - 최종
    ```cpp
    class Solution {
    private:
        int dx[3] = {1,0,1};
        int dy[3] = {0,1,1};
    public:
        int maximalSquare(vector<vector<char>>& matrix) {
            int m = matrix.size();
            int n = matrix[0].size();
            int dp[301][301] = {0};

            int ret = 0;
            for(int i = 0;i<m;i++){
                for(int j = 0;j<n;j++){
                    if(matrix[i][j] == '1'){
                        if(i == 0 || j == 0){
                            dp[i][j] = 1;
                        }
                        else{
                            dp[i][j] = min({dp[i-1][j],dp[i-1][j-1],dp[i][j-1]}) + 1;
                        }
                        ret = max(ret, dp[i][j]);
                    }
                }
            }
            return ret*ret;
        }
    };
    ```

    - 사이즈별 접근
    ```cpp
    class Solution {
    private:
        int dx[3] = {1,0,1};
        int dy[3] = {0,1,1};
    public:
        int maximalSquare(vector<vector<char>>& matrix) {
            int m = matrix.size();
            int n = matrix[0].size();
            int maxSquare = min(m, n);
            bool dp[301][301] = {false};

            // Init
            bool flag = false;
            for(int x = 0;x<m;x++){
                for(int y = 0;y<n;y++){
                    if(matrix[x][y] == '1'){
                        dp[x][y] = true;
                        flag = true;
                    }
                }
            }
            if(flag == false) return 0;

            int size = 2;
            for(;size <= maxSquare; size++){
                flag = false;
                for(int x = 0;x<m;x++){
                    for(int y = 0;y<n;y++){
                        if(dp[x][y] == false) continue;
                        int res = 0;
                        for(int i = 0;i<3;i++){
                            int nx = x + dx[i];
                            int ny = y + dy[i];
                            if(nx>=m || ny >= n) continue;
                            if(dp[nx][ny]) res++;
                        }
                        if(res == 3){
                            dp[x][y] = true;
                            flag = true;
                        }else{
                            dp[x][y] = false;
                        }
                    }
                }
                if(flag == false) return (size-1)*(size-1);
            }

            return maxSquare*maxSquare;
        }
    };
    ```
