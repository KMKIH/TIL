### 코딩 테스트
- Course Schedule
    - 문제 접근
        - 선행조건을 따져야하는 문제
        - 위상정렬이 바로 떠올랐다
    - 구현
        ```cpp
        class Solution {
        public:
            bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
                // 위상 정렬
                vector<int> indegree(numCourses,0);
                vector<vector<int>> adj(numCourses,vector<int>());

                for(auto& prereq : prerequisites){
                    int a = prereq[0];
                    int b = prereq[1];

                    indegree[a]++;
                    adj[b].push_back(a);
                }

                int count = 0;
                queue<int> q;
                for(int i = 0;i<numCourses;i++){
                    if(indegree[i] == 0){
                        count++;
                        q.push(i);
                    }
                }
                
                while(q.empty() == false){
                    int cur = q.front();q.pop();
                    for(int i = 0;i<adj[cur].size();i++){
                        int next = adj[cur][i];
                        indegree[next]--;
                        if(indegree[next] == 0) {
                            count++;
                            q.push(next);
                        }
                    }
                }

                return count == numCourses;
            }
        };
        ```
