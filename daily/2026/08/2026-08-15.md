## 정리
### 코딩 테스트
- Best Team With No Conflicts
    > 0. dp 부터 생각하지 말고 재귀부터 생각한다
    
    - 문제 정리
        1. 나이가 어린순서부터 접근한다
        2. 최고 점수를 기록하여, 다음 팀원을 선택을 할 때 conflict가 있는지 확인한다
    
    > 1. 상태 정의
    
    - solve(i) = i번째 선수까지 살펴보았을 때 최고 점수

    > 2. 현재 상태에서 가능한 선택

    - 다음 선수 중 conflict가 없는 선수 선택

    > 4. 점화식
    for(int i = idx + 1; i < n ; i++){
        solve(idx) = max(solve(idx), solve(i));
    }
    - 재귀 -> 메모이제이션
        ```cpp
            class Solution {
            private:
                vector<pair<int,int>> info;
                vector<int> cache;
                int solve(int idx, int maxScore){
                    int& ret = cache[idx];
                    if(ret != -1) return ret;
                    ret = 0;

                    for(int i = idx+1;i< info.size();i++){
                        if(info[i].second < maxScore) continue;
                        ret = max(ret, solve(i, info[i].second)+info[i].second);
                    }
                    return ret;
                }
            public:
                int bestTeamScore(vector<int>& scores, vector<int>& ages) {
                    int n = scores.size();
                    int maxScore = 0;

                    info = vector<pair<int,int>>(n+1);
                    for(int i = 1;i<=n;i++){
                        info[i] =  {ages[i-1],scores[i-1]};
                        maxScore = max(maxScore,scores[i-1]);
                    }
                    sort(info.begin()+1,info.end());

                    cache = vector<int>(n+1,-1);
                    return solve(0, 0);
                }
            };
        ```
    - DP
        ```cpp
        class Solution {
        public:
            int bestTeamScore(vector<int>& scores, vector<int>& ages) {
                int n = scores.size();

                vector<pair<int,int>> info = vector<pair<int,int>>(n);
                vector<int> dp = vector<int>(n,-1);
                
                info = vector<pair<int,int>>(n);
                for(int i = 0;i<n;i++){
                    info[i] =  {ages[i],scores[i]};
                }
                sort(info.begin(),info.end());

                int ret = 0;
                for(int i = 0;i<n;i++){
                    dp[i] = info[i].second;
                    for(int j = 0;j<i;j++){
                        if(info[j].second <= info[i].second){
                            dp[i] = max(dp[i], dp[j] + info[i].second);
                        }
                    }
                    ret = max(ret, dp[i]);
                }
                return ret;
            }
        };
        ```
