### 코딩 테스트
- Combination Sum
    - 문제 접근
        - 모든 경우를 시도해보자 (백트래킹)
            1. 요소를 선택한다
                - 중복 선택이 가능하므로, 자기 자신부터 선택할 요소를 고른다
                - 이때 같은 조합의 중복생성을 막기 위해, 원소의 순서를 강제한다
            2. 그 합이 target과 같다면 정답 배열에 추가한다
    - 구현
        ```cpp
            class Solution {
            private:
                vector<vector<int>> ret;
                void rec(vector<int>& res, vector<int>& candidates, int idx, int sum, int target){
                    if(sum == target){
                        ret.push_back(res);
                        return;
                    }
                    if(sum > target) return;

                    for(int i = idx;i<candidates.size();i++){
                        res.push_back(candidates[i]);
                        rec(res,candidates,i,sum + candidates[i], target);
                        res.pop_back();
                    }
                }
            public:
                vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
                    vector<int> res;
                    rec(res, candidates, 0, 0, target);
                    return ret;
                }
            };
        ```
    - 다른 접근
        > 문제 풀이 이후 다른 사람의 풀이도 보았다
        - 기존 나의 아이디어는 현재 위치에서 자신을 포함한 이후의 요소들을 선택하는 방식이다
        - 다른 아이디어는 현재 위치의 요소를 선택할지 말지를 결정하는 방식이다 
    - 구현
        ```cpp
        class Solution {
        private:
            vector<vector<int>> ret;
            void rec(vector<int>& res, vector<int>& candidates, int idx, int sum, int target){
                if(sum == target){
                    ret.push_back(res);
                    return;
                }
                if(sum > target || idx >= candidates.size()) return;

                // idx 선택한다
                res.push_back(candidates[idx]);
                rec(res,candidates,idx,sum + candidates[idx], target);
                res.pop_back();

                // idx를 선택하지 않는다
                rec(res,candidates,idx+1,sum, target);
            }
        public:
            vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
                vector<int> res;
                rec(res, candidates, 0, 0, target);
                return ret;
            }
        };
        ```
