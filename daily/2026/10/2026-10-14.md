### 코딩 테스트

- Sum of Subarray Minimums (풀이 실패)
    - 문제 요약
        - 모든 연속된 부분 수열들에서 제일 작은 원소의 합을 구하자
    ---
    - 1차 시도
        - 고정된 슬라이딩 윈도우로 크기별로 순회하자
        - 최소 값을 유지하기 위해서는 단조 큐를 이용하자
    - 한계
        - 고정 크기별로 계산하게되면, 최소값 연산이 여러번하게 되므로, 고정크기로 하면 안되겠다
    ---
    - 2차 시도
        - 투포인터를 통한 모든 경우 순회
    - 의도
        - 최소값 연산을 최소화하자
    - 한계
        - BruteForce가 되어 시간 초과...
    - 코드
        ```cpp
        class Solution {
        public:
            int sumSubarrayMins(vector<int>& arr) {
                int ret = 0;
                const int mod = 1000000007;
                // Window 초기 크기 = 1
                // 일단 r을 증가시킨다
                // r이 더이상 늘어날 수 없을때, l을 증가시키고 r을 l과 같게한다
                // l과 r을 이동하는 과정에서 최소값을 갱신하자
                
                for(int l = 0; l < arr.size();l++){
                    queue<int> monoQ;
                    monoQ.push(arr[l]);
                    
                    for(int r = l; r < arr.size();r++){
                        while(monoQ.empty()==false && monoQ.back() > arr[r]) monoQ.pop();
                        monoQ.push(arr[r]);

                        ret += monoQ.front();
                        ret %= mod;
                    }
                }
                return ret;
            }
        };
        ```
    ---
    - 3차 시도
        - 관점을 바꾸어보자
        - 하나의 수에 대해서 그 값을 최소로 하는 수열들을 계산하자
    - 한계
        1. 동일한 수에 대한 처리가 되어있지 않다
            - 등호를 사용하면 동일한 부분 수열을 계산하고
            - 등호를 사용하지 않으면 놓치는 부분 수열이 생긴다
        2. 여전히 시간 초과
    - 코드
        ```cpp
            class Solution {
            public:
                int sumSubarrayMins(vector<int>& arr) {
                    long long ret = 0;
                    const int mod = 1000000007;
                    
                    // 다른 관점으로 접근해보자
                    // 1. 하나의 수를 최소값으로 생각했을때 만들 수 있는 배열의 갯수

                    for(int i = 0; i< arr.size();i++){
                        int minNum = arr[i];
                        int l = i-1;
                        int r = i+1;

                        while(l >= 0 && arr[l] > minNum)l--;
                        l++;
                        while(r < arr.size() && arr[r] > minNum) r++;
                        r--;

                        ret += (r-l+1) * (long long)minNum;
                        ret %= mod;
                        ret += ((i-l)*(r-i)) * (long long)minNum;
                        ret %= mod;
                    }
                    return ret;
                }
            };
        ```
    --- 
    - 4차 시도
        - 중복 연산 처리
    - 방법
        - visited를 사용해, 이미 처리한 수에 대해서는 처리하지 않는다
    - 한계
        - 여전히 시간 초과
    - 코드
        ```cpp
        class Solution {
        public:
            int sumSubarrayMins(vector<int>& arr) {
                long long ret = 0;
                const int mod = 1000000007;
                vector<bool> visited(arr.size(),false);
                
                // 다른 관점으로 접근해보자
                // 1. 하나의 수를 최소값으로 생각했을때 만들 수 있는 배열의 갯수

                for(int i = 0; i< arr.size();i++){
                    int l = i-1;
                    int r = i+1;

                    while(l >= 0 && (arr[l] > arr[i]|| (arr[l] == arr[i] && visited[l]==false)))l--;
                    l++;
                    while(r < arr.size() && (arr[r] > arr[i]|| (arr[r] == arr[i] && visited[r]==false))) r++;
                    r--;

                    ret += (r-l+1) * (long long)arr[i];
                    ret %= mod;
                    ret += ((i-l)*(r-i)) * (long long)arr[i];
                    ret %= mod;
                    visited[i] = true;
                }
                return ret;
            }
        };
        ```
    ---
    - 힌트
        - 현재 코드에서 중복 연산이 발생하는 부분은 경계를 찾는 부분이다
        - 해당 위치를 미리 계산하여 중복 연산을 없애자

- Final Prices With a Special Discount in a Shop
    ```cpp
    class Solution {
    public:
        vector<int> finalPrices(vector<int>& prices) {
            // 1. i의 오른쪽에 위치
            // 2. prices[i]보다 작거나 같아야한다

            // 오른쪽에서 부터 보자
            
            vector<int> ret;
            stack<int> s;
            for(int i = prices.size()-1; i>=0; i--){
                // 필요한 데이터
                // 1. 나보다 작거나 같은 값들의 목록
                // 2. 스택 내에서 내림차순이 되는 순간 그 값은 의미가 없어진다
                    // => 스택은 오름차순을 유지한다

                // 스택 갱신 (나보다 큰 값들을 제거한다)
                while(s.empty()==false && s.top() > prices[i]) s.pop();

                // 결과
                ret.push_back(prices[i]-(s.empty()?0:s.top()));

                // 스택 갱신 (나를 넣는다)
                s.push(prices[i]);
            }
            reverse(ret.begin(), ret.end());
            return ret;
        }
    };
    ```
